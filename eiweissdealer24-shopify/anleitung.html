Produkte importieren

Die Datei produkte-import.csv hat dir Claude im Chat geschickt. Speichere sie zuerst (iPad: Datei antippen → Teilen → In Dateien sichern).

  1. admin.shopify.com → Mein Shop → Produkte → Importieren
  2. Datei hinzufügen → produkte-import.csv auswählen
  3. Hochladen und Vorschau → Produkte importieren

Shopify lädt dabei die Produktbilder und schickt dir eine E-Mail, wenn alles fertig ist (ein paar Minuten). Alle Sorten und Größen eines Produkts haben den „ab“-Preis aus dem Original-Shop, denn die Preise einzelner Größen stehen dort nicht. Für die Präsentation passt das. Bevor echte Bestellungen möglich sind, müssen die Preise größerer Packungen nachgetragen werden. Der Lagerbestand wird nicht verfolgt, also ist nichts „ausverkauft“.

Arbeite an einer Kopie, dann bleibt dein aktuelles Theme unverändert, bis alles fertig ist.

  1. Onlineshop → Themes → beim aktuellen Theme auf … → Duplizieren
  2. Bei der Kopie auf … → Code bearbeiten

9 Abschnitte einfügen

Für jede Datei unten wiederholst du diese vier Handgriffe:

  1. Im Code-Editor links den Ordner sections öffnen → + Neuen Abschnitt hinzufügen (bzw. „Datei hinzufügen“)
  2. Als Namen genau den Dateinamen eingeben, z. B. ed24-hero, ohne Endung. Fragt Shopify nach dem Typ, Liquid wählen → Fertig
  3. Den vorgegebenen Inhalt komplett löschen (alles markieren → löschen), dann hier Code kopieren tippen und dort einfügen
  4. Speichern → hier Eingefügt abhaken

Wichtig: Den Dateinamen exakt so schreiben wie hier angegeben, sonst finden sich die Abschnitte nicht gegenseitig. Eine Fehlermeldung beim Speichern heißt fast immer: Es wurde nicht der komplette Code eingefügt. Dann noch einmal alles löschen und neu einfügen.

  // Produkt-Showcase: zeigt das Kapitel passend zur Scroll-Position; im Theme-Editor springt die Ansicht zum gewählten Kapitel.
  var ed24Clamp = function (v) { return Math.min(1, Math.max(0, v)); };
  class Ed24Showcase extends HTMLElement {
    connectedCallback() {
      var self = this;
      this.ctrl = new AbortController();
      var opts = { signal: this.ctrl.signal, passive: true };
      this.section = this.closest('[data-ed24-showcase]') || this;
      this.chapters = Array.prototype.slice.call(this.querySelectorAll('[data-ed24-chapter]'));
      this.num = this.querySelector('[data-ed24-num]');
      this.bar = this.querySelector('[data-ed24-bar]');
      this.active = -2;
      this.chapters.forEach(function (el) { el.inert = true; });
      var queue = function () {
        if (self.tick) return;
        self.tick = requestAnimationFrame(function () { self.tick = 0; self.update(); });
      };
      window.addEventListener('scroll', queue, opts);
      window.addEventListener('resize', queue, opts);
      document.addEventListener('shopify:block:select', function (e) {
        var i = self.chapters.findIndex(function (el) { return el.dataset.blockId === (e.detail && e.detail.blockId); });
        if (i < 0) return;
        var top = self.section.getBoundingClientRect().top + window.scrollY;
        var span = Math.max(1, self.section.offsetHeight - window.innerHeight);
        window.scrollTo({ top: top + span * ((i + 0.5) / self.chapters.length), behavior: 'auto' });
      }, { signal: this.ctrl.signal });
      this.update();
    }

    disconnectedCallback() {
      if (this.ctrl) this.ctrl.abort();
      cancelAnimationFrame(this.tick);
      this.tick = 0;
    }

    update() {
      var n = this.chapters.length;
      if (!n) return;
      var r = this.section.getBoundingClientRect();
      var span = Math.max(1, r.height - window.innerHeight);
      var progress = ed24Clamp(-r.top / span);
      var current = r.top > window.innerHeight * 0.35 ? -1 : Math.min(n - 1, Math.floor(progress * n + 1e-6));
      if (current === this.active) return;
      this.active = current;
      this.chapters.forEach(function (el, i) {
        el.classList.toggle('is-on', i === current);
        el.classList.toggle('is-past', current > -1 && i < current);
        el.inert = i !== current;
      });
      if (current < 0) return;
      if (this.num) this.num.textContent = (current + 1 < 10 ? '0' : '') + (current + 1);
      if (this.bar) this.bar.style.transform = 'scaleX(' + (current + 1) / n + ')';
    }
  }
  if (!customElements.get('ed24-showcase')) customElements.define('ed24-showcase', Ed24Showcase);

        if (self.tick) return;

      window.addEventListener('resize', queue, opts);

  // Laufschrift: läuft dauerhaft und wird mit der Scroll-Geschwindigkeit schneller.
  class Ed24Marquee extends HTMLElement {
    connectedCallback() {
      var self = this;
      this.track = this.querySelector('[data-ed24-track]');
      if (!this.track || window.matchMedia('(prefers-reduced-motion: reduce)').matches || !('IntersectionObserver' in window)) return;
      this.x = 0;
      this.lastY = window.scrollY;
      this.last = performance.now();
      this.observer = new IntersectionObserver(function (entries) {
        cancelAnimationFrame(self.raf);
        if (entries[0].isIntersecting) {
          self.half = self.track.scrollWidth / 2 || 1;
          self.last = performance.now();
          self.raf = requestAnimationFrame(function (t) { self.frame(t); });
        }
      });
      this.observer.observe(this);
    }

    disconnectedCallback() {
      if (this.observer) this.observer.disconnect();
      cancelAnimationFrame(this.raf);
    }

    frame(now) {
      var self = this;
      this.raf = requestAnimationFrame(function (t) { self.frame(t); });
      var dt = Math.min(0.05, (now - this.last) / 1000);
      this.last = now;
      var v = Math.abs(window.scrollY - this.lastY) / Math.max(dt, 0.001);
      this.lastY = window.scrollY;
      this.x = (this.x + (40 + Math.min(v, 4000) * 0.08) * dt) % this.half;
      this.track.style.transform = 'translate3d(' + -this.x + 'px,0,0)';
    }
  }
  if (!customElements.get('ed24-marquee')) customElements.define('ed24-marquee', Ed24Marquee);

Startseite zusammenbauen

Bei der Theme-Kopie auf Anpassen tippen. Oben ist die Startseite ausgewählt.

  1. Vorhandene Abschnitte der Startseite, die du nicht brauchst (z. B. Bild-Banner, Diashow), ausblenden (Auge-Symbol) oder löschen
  2. In der Kopfzeile (Header-Bereich oben) Abschnitt hinzufügen → ED24 Topbar
  3. In der Vorlage (Startseite) die Abschnitte in dieser Reihenfolge hinzufügen und mit dem Griff ⠿ sortieren:
  1. ED24 Hero 3D ganz oben
  2. ED24 Produkt-Showcase direkt darunter
  3. ED24 Story
  4. ED24 Coaching-Panel hier endet die 3D-Bühne
  5. Theme-Abschnitt Ausgewählte Kollektion (bei neueren Themes „Produktliste“) z. B. „Eigenmarke“ oder „Angebote“
  6. ED24 Ziele
  7. ED24 Marken (Physik)
  8. ED24 Laden & Team
  9. ED24 Laufschrift als letzter Abschnitt

Reihenfolge 1–4 nicht unterbrechen. Hero, Showcase, Story und Coaching-Panel teilen sich eine 3D-Bühne. Ein anderer Abschnitt dazwischen würde von ihr überdeckt.
